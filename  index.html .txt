  <script>
    // 🔴 CHANGE THIS PASSWORD TO YOUR OWN!
    const MASTER_KEY = 'doctor123';
    
    if (localStorage.getItem('adminSession') === 'active') {
      showAdmin();
    }

    function authenticate() {
      const input = document.getElementById('masterKey').value;
      if (input === MASTER_KEY) {
        localStorage.setItem('adminSession', 'active');
        showAdmin();
      } else {
        const err = document.getElementById('errorMsg');
        err.style.display = 'block';
        setTimeout(() => err.style.display = 'none', 2000);
      }
    }

    function showAdmin() {
      document.getElementById('loginScreen').classList.add('hidden');
      document.getElementById('adminPanel').classList.remove('hidden');
    }

    function logout() {
      localStorage.removeItem('adminSession');
      location.reload();
    }

    function generateLink() {
      const id = Math.random().toString(36).substring(7);
      const link = `https://your-doctor-site.com/patient/${id}`;
      
      const temp = document.createElement('input');
      temp.value = link;
      document.body.appendChild(temp);
      temp.select();
      document.execCommand('copy');
      document.body.removeChild(temp);
      
      let count = parseInt(document.getElementById('linkCount').textContent);
      document.getElementById('linkCount').textContent = count + 1;
      
      alert('Link copied: ' + link);
    }

    setTimeout(() => {
      if (localStorage.getItem('adminSession')) logout();
    }, 30 * 60 * 1000);
  </script>
